\begin{theorem}[Nested pinned-span ladder at error rank eleven]
\label{thm:mca-rank11-nested-pinned-span}
Retain the anchored rich-flat hypotheses at the deployed KoalaBear row.
At cutoff $\tau=1937$ and threshold $h=36775$, every over-budget line
admits nested actual coordinate sets
\[
 T_1\subset\cdots\subset T_{10},\qquad |T_k|=k,
\]
and nested direction spaces
\[
 V_{10}\le\cdots\le V_1\le C'
\]
such that $T_k$ is a common-zero set of $V_k$.  The assigned distinct-slope
loads for $k=1,\ldots,10$ are at least
\[
\begin{gathered}
2843853816476423,\ 93708171878891,\ 3087708134499,\\
101738094101,\ 3352119806,\ 110444488,\ 3638792,\ 119884,\ 3950,\ 131.
\end{gathered}
\]
Moreover,
\[
 \dim V_1\ge8,\quad \dim V_2\ge7,\quad \dim V_3\ge6,\quad
 \dim V_4\ge5,\quad \dim V_5\ge3,\quad \dim V_k\ge2\ (6\le k\le10).
\]
\end{theorem}
